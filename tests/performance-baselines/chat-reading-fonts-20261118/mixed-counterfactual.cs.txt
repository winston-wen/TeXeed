using System.Diagnostics;
using System.IO;
using System.Security.Cryptography;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Documents;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.Controls;
using TeXeed.App.Services;

internal static class ChatReadingFontPerf
{
    private static async Task Idle(Window window) { await window.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); window.UpdateLayout(); }
    private static async Task Drain(Window window) { await Idle(window); await Task.Delay(800); await Idle(window); }
    private static IEnumerable<T> Children<T>(DependencyObject root) where T : DependencyObject
    {
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        { var child = VisualTreeHelper.GetChild(root, i); if (child is T match) yield return match; foreach (var inner in Children<T>(child)) yield return inner; }
    }
    private static int Visible(Window window)
    {
        var bounds = new Rect(0, 0, window.ActualWidth, window.ActualHeight);
        return Children<Control>(window).Count(control =>
        {
            if (!control.IsVisible || control.ActualWidth <= 0 || control.ActualHeight <= 0) return false;
            try { return control.TransformToAncestor(window).TransformBounds(new Rect(0, 0, control.ActualWidth, control.ActualHeight)).IntersectsWith(bounds); }
            catch (InvalidOperationException) { return false; }
        });
    }
    private static IEnumerable<GlyphRun> GlyphRuns(Drawing? drawing)
    {
        if (drawing is GlyphRunDrawing glyph) yield return glyph.GlyphRun;
        else if (drawing is DrawingGroup group)
            foreach (var child in group.Children)
                foreach (var run in GlyphRuns(child)) yield return run;
    }
    private static GlyphRun[] DrawRuns(FontFamily family, bool dottedZero)
    {
        var block = new TextBlock { Text = "Il1 O0 中文", FontFamily = family, FontSize = 40 };
        Typography.SetStylisticSet4(block, dottedZero);
        block.Measure(new Size(900, 120)); block.Arrange(new Rect(0, 0, 900, 120)); block.UpdateLayout();
        return GlyphRuns(VisualTreeHelper.GetDrawing(block)).ToArray();
    }
    [STAThread] private static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); var seeds = Path.GetFullPath(args[1]); var oldFont = args.Length > 2 && args[2] == "old-font"; Directory.CreateDirectory(output);
        var home = Path.Combine(output, "profile"); var cache = Path.Combine(home, "ai", "math-svg"); Directory.CreateDirectory(cache);
        foreach (var file in Directory.EnumerateFiles(seeds, "*.svg")) File.Copy(file, Path.Combine(cache, Path.GetFileName(file)), true);
        Environment.SetEnvironmentVariable("TEXEED_HOME", home);
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        ThemeManager.Apply("ristretto");
        const string equation = @"\operatorname{Adv}^{\mathrm{dMLWR}}_{D} = \left|\Pr[D(A,s)=1]-\Pr[D(A,u)=1]\right|+\Pr[D(A,t)=0]";
        var paragraph = string.Join("\n\n", Enumerable.Range(0, 30).Select(i => $"第 {i} 段。接收方核对 Il1 O0 Key 1024；输出 $0$ 或 $1$，由算法 $D$ 检查。"));
        var text = paragraph + "\n\n$$" + equation + "$$";
        var view = new MarkdownView { ChatMode = true };
        var host = new Window { Width = 900, Height = 760, Left = -12000, Top = 20, ShowActivated = false, ShowInTaskbar = false, WindowStartupLocation = WindowStartupLocation.Manual, Content = view };
        var result = 1;
        app.DispatcherUnhandledException += (_, e) => { File.WriteAllText(Path.Combine(output, "error.txt"), e.Exception.ToString()); e.Handled = true; app.Shutdown(1); };
        host.Loaded += async (_, _) =>
        {
            try
            {
                view.Markdown = text;
                async Task Settle()
                {
                    await view.RenderTask; await Idle(host);
                    var formulas = Children<ChatFormula>(view).Select(formula => formula.RenderTask).ToArray();
                    await Task.WhenAll(formulas).WaitAsync(TimeSpan.FromSeconds(30)); await Drain(host);
                }
                await Settle();
                var typography = typeof(MarkdownView).Assembly.GetType("TeXeed.App.Controls.ChatReadingTypography");
                object fontAudit = new { revisedTypography = false };
                if (typography is not null)
                {
                    var local = (string)typography.GetProperty("LocalFontDirectory", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic)!.GetValue(null)!;
                    var hiragino = Path.Combine(local, "Hiragino_Sans_GB.ttc");
                    var ibm = Path.Combine(AppContext.BaseDirectory, "Assets", "Fonts", "IBMPlexSans-Regular.otf");
                    if (!File.Exists(hiragino) || !File.Exists(ibm)) throw new InvalidOperationException("Required reading fonts are not available to this process");
                    var runs = DrawRuns(view.FontFamily, true); var ordinary = DrawRuns(view.FontFamily, false);
                    var latin = runs.FirstOrDefault(run => run.GlyphTypeface.FontUri.LocalPath.EndsWith("IBMPlexSans-Regular.otf", StringComparison.OrdinalIgnoreCase));
                    var latinWithoutSs04 = ordinary.FirstOrDefault(run => run.GlyphTypeface.FontUri.LocalPath.EndsWith("IBMPlexSans-Regular.otf", StringComparison.OrdinalIgnoreCase));
                    var chinese = runs.FirstOrDefault(run => run.GlyphTypeface.FontUri.LocalPath.EndsWith("Hiragino_Sans_GB.ttc", StringComparison.OrdinalIgnoreCase));
                    var matchesLatin = latin is not null && latinWithoutSs04 is not null;
                    var matchesChinese = chinese is not null && chinese.GlyphTypeface.FaceNames.Values.Contains("W3", StringComparer.OrdinalIgnoreCase);
                    var matchesSs04 = matchesLatin && latin!.GlyphIndices.Count == latinWithoutSs04!.GlyphIndices.Count && latin.GlyphIndices.Zip(latinWithoutSs04.GlyphIndices).Count(pair => pair.First != pair.Second) == 1;
                    var actual = runs.Select(run => new { file = Path.GetFileName(run.GlyphTypeface.FontUri.LocalPath), face = string.Join(",", run.GlyphTypeface.FaceNames.Values.Distinct()) }).ToArray();
                    if (!matchesLatin || !matchesChinese || !matchesSs04) throw new InvalidOperationException("Reading glyph check failed: " + JsonSerializer.Serialize(new { matchesLatin, matchesChinese, matchesSs04, actual }));
                    fontAudit = new { revisedTypography = true, hiraginoAvailable = true, hiraginoSha256 = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(hiragino))), ibmSha256 = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(ibm))), latinGlyphsMatchIbm = matchesLatin, chineseGlyphsMatchHiragino = matchesChinese, dottedZeroSs04 = matchesSs04, actual };
                }
                if (oldFont) view.FontFamily = new FontFamily("Times New Roman, Source Han Serif SC, FangSong");
                view.Markdown = text + "\n\n预热。"; await Settle();
                var samples = new List<object>();
                async Task Measure(string operation, Action<int> action)
                {
                    for (var i = 0; i < 9; i++)
                    {
                        var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(host); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                        var bytes = GC.GetAllocatedBytesForCurrentThread(); var timer = Stopwatch.StartNew(); action(i); await Idle(host);
                        var feedback = timer.Elapsed.TotalMilliseconds; var feedbackBytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                        await Settle(); timer.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                        var controls = Visible(host); var formulas = Children<ChatFormula>(view).Count();
                        if (!host.IsVisible || controls == 0 || formulas < 90) throw new InvalidOperationException("Typography fixture lost its window or formula visuals");
                        var usesNewFont = view.Document.FontFamily.Source.Contains("IBM Plex Sans", StringComparison.Ordinal);
                        if (oldFont == usesNewFont) throw new InvalidOperationException("The intended font counterfactual was not retained during rendering");
                        if (i >= 2) samples.Add(new { op = operation, feedbackMs = feedback, feedbackBytes, rawMs = timer.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = timer.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = controls, formulaVisuals = formulas });
                    }
                }
                await Measure("mixed-render", i => view.Markdown = text + (i % 2 == 0 ? "\n\n末句甲。" : "\n\n末句乙。"));
                await Measure("mixed-resize", i => host.Width = i % 2 == 0 ? 620 : 900);
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "Counterfactual: same new assembly and host with actual reading font or temporary old-font override; 30 mixed paragraphs, 90 inline formulas and one display formula, cached SVG; 2 warmups, 7 samples, paired 800ms idle", oldFont, fontAudit, samples }, new JsonSerializerOptions { WriteIndented = true }));
                result = 0;
            }
            catch (Exception ex) { File.WriteAllText(Path.Combine(output, "error.txt"), ex.ToString()); }
            finally { host.Close(); app.Shutdown(result); }
        };
        host.Show(); app.Run(); return result;
    }
}
