static async Task SettleThumbnails(DependencyObject? root,Window host){
 var tasks=new List<Task>();
 void Walk(DependencyObject? node){if(node is null)return;if(node.GetType().Name=="AttachmentThumbnail" && node.GetType().GetProperty("LoadTask",BindingFlags.Instance|BindingFlags.NonPublic)?.GetValue(node) is Task task)tasks.Add(task);for(var i=0;i<VisualTreeHelper.GetChildrenCount(node);i++)Walk(VisualTreeHelper.GetChild(node,i));}
 Walk(root);try{await Task.WhenAll(tasks);}catch(OperationCanceledException){}await Idle(host);
}
static int VisibleImages(DependencyObject? root){if(root is null)return 0;var count=root is Image img&&img.IsVisible&&img.Source is not null?1:0;for(var i=0;i<VisualTreeHelper.GetChildrenCount(root);i++)count+=VisibleImages(VisualTreeHelper.GetChild(root,i));return count;}
static int Visible(DependencyObject root){
 var size=root is FrameworkElement r?new Rect(0,0,r.ActualWidth,r.ActualHeight):Rect.Empty;int count=0;
 void Walk(DependencyObject node){for(int i=0;i<VisualTreeHelper.GetChildrenCount(node);i++){var child=VisualTreeHelper.GetChild(node,i);if(child is Control c&&c.IsVisible&&c.ActualWidth>0&&c.ActualHeight>0){try{var rect=c.TransformToAncestor((Visual)root).TransformBounds(new Rect(0,0,c.ActualWidth,c.ActualHeight));rect.Intersect(size);for(DependencyObject? p=VisualTreeHelper.GetParent(c);p is not null&&p!=root;p=VisualTreeHelper.GetParent(p)){if(p is FrameworkElement f&&(f.ClipToBounds||f is ScrollViewer||f is ScrollContentPresenter)){var clip=f.TransformToAncestor((Visual)root).TransformBounds(new Rect(0,0,f.ActualWidth,f.ActualHeight));rect.Intersect(clip);}}if(!rect.IsEmpty&&rect.Width>0&&rect.Height>0)count++;}catch(InvalidOperationException){}}Walk(child);}}Walk(root);return count;}
