using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

internal static class MarkExitComplete
{
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private static object Field(object obj, string name) => obj.GetType().GetField(name, Private)!.GetValue(obj)!;
    private static void Mode(AiPanel panel, bool value) => typeof(AiPanel).GetMethod("SetAnswerMarking", Private)!.Invoke(panel, [value]);
    private static async Task Idle(Window window) { await window.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); window.UpdateLayout(); }
    private static async Task Drain(AiPanel panel, Window window)
    {
        await Task.Delay(3500);
        var timer = (DispatcherTimer)Field(panel, "_saveTimer");
        for (var i = 0; timer.IsEnabled && i < 250; i++) await Task.Delay(20);
        if (timer.IsEnabled) throw new TimeoutException("The draft save timer did not drain.");
        var gate = (SemaphoreSlim)Field(panel, "_saveGate"); await gate.WaitAsync(); gate.Release(); await Idle(window);
    }
    private static int Visible(DependencyObject root, Window host)
    {
        var count = 0;
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        {
            var child = VisualTreeHelper.GetChild(root, i);
            if (child is Control c && c.IsVisible && c.ActualWidth > 0 && c.ActualHeight > 0)
                try { if (c.TransformToAncestor(host).TransformBounds(new Rect(0, 0, c.ActualWidth, c.ActualHeight)).IntersectsWith(new Rect(0, 0, host.ActualWidth, host.ActualHeight))) count++; } catch (InvalidOperationException) { }
            count += Visible(child, host);
        }
        return count;
    }
    [STAThread] private static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
        Environment.SetEnvironmentVariable("TEXEED_HOME", Path.Combine(output, "profile"));
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        var panel = new AiPanel(); typeof(AiPanel).GetMethod("InitializeBranchUi", Private)?.Invoke(panel, null);
        var host = new Window { Width = 900, Height = 760, Left = -12000, Top = 20, ShowActivated = false, ShowInTaskbar = false, WindowStartupLocation = WindowStartupLocation.Manual, Content = panel };
        var exit = 1; app.DispatcherUnhandledException += (_, e) => { Console.Error.WriteLine(e.Exception); e.Handled = true; app.Shutdown(1); };
        host.Loaded += async (_, _) =>
        {
            try
            {
                await panel.SetProjectAsync(Path.Combine(output, "fixture.tex")); var book = (AiNotebook)Field(panel, "_book");
                for (var i = 0; i < 120; i++) book.Turns.Add(new AiTurn("如何检查结果？ " + i, "样例回答：说明步骤和依据。") { SentAt = DateTimeOffset.UnixEpoch, Segments = Enumerable.Range(0, 3).Select(k => new AiSegment { Kind = "activity", Text = "检查步骤 " + k, Details = string.Join('\n', Enumerable.Range(0, 200).Select(n => "示例日志 " + n)) }).Append(new AiSegment { Kind = "final", Text = "样例回答：说明步骤和依据。" }).ToList() });
                typeof(AiPanel).GetMethod("RefreshTranscript", Private)!.Invoke(panel, null); await Idle(host);
                var status = (TextBlock)panel.FindName("StatusText"); var samples = new List<object>();
                for (var i = 0; i < 9; i++)
                {
                    Mode(panel, true); await Idle(host);
                    var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(panel, host); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                    var bytes = GC.GetAllocatedBytesForCurrentThread(); var sw = Stopwatch.StartNew(); Mode(panel, false); await Idle(host); var feedback = sw.Elapsed.TotalMilliseconds; var immediate = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    var immediateText = status.Text; var immediateHeight = status.IsVisible ? status.ActualHeight : 0;
                    await Drain(panel, host); sw.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    if (!host.IsVisible || Visible(host, host) == 0) throw new InvalidOperationException("The benchmark window closed before sampling completed.");
                    if (i >= 2) samples.Add(new { op = "exit-mark-complete", feedbackMs = feedback, feedbackBytes = immediate, rawMs = sw.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = sw.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = Visible(host, host), immediateText, immediateHeight, finalText = status.Text, finalHeight = status.IsVisible ? status.ActualHeight : 0 });
                }
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "120 turns, 900x760 offscreen WPF; 2 warmups and 7 samples, paired 3500ms, save gate and ApplicationIdle; includes 3-second hint cleanup", samples }, new JsonSerializerOptions { WriteIndented = true }));
                exit = 0;
            }
            catch (Exception ex) { Console.Error.WriteLine(ex); }
            finally { panel.Dispose(); host.Close(); app.Shutdown(exit); }
        };
        host.Show(); app.Run(); return exit;
    }
}
