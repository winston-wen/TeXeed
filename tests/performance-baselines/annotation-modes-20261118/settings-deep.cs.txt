using System.Diagnostics;
using System.IO;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App;
using TeXeed.Core;

internal static class SettingsDeep
{
    private static async Task Idle(Window w) { await w.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); w.UpdateLayout(); }
    private static async Task Drain(Window w) { await Idle(w); await Task.Delay(250); await Idle(w); }
    private static int Visible(DependencyObject root)
    {
        var count = 0;
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        { var child = VisualTreeHelper.GetChild(root, i); if (child is Control c && c.IsVisible && c.ActualWidth > 0 && c.ActualHeight > 0) count++; count += Visible(child); }
        return count;
    }
    [STAThread] private static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output); Environment.SetEnvironmentVariable("TEXEED_HOME", Path.Combine(output, "profile"));
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown }; app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        var sentinel = new Window { Width = 1, Height = 1, Left = -12000, Top = 20, ShowActivated = false, WindowStartupLocation = WindowStartupLocation.Manual }; var exit = 1;
        app.DispatcherUnhandledException += (_, e) => { Console.Error.WriteLine(e.Exception); e.Handled = true; app.Shutdown(1); };
        sentinel.Loaded += async (_, _) =>
        {
            try
            {
                var samples = new List<object>(); await Drain(sentinel);
                for (var i = 0; i < 9; i++)
                {
                    var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(sentinel); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                    var bytes = GC.GetAllocatedBytesForCurrentThread(); var watch = Stopwatch.StartNew();
                    var dialog = new SettingsWindow(new AppSettings()) { Width = 760, Height = 700, Left = -12000, Top = 20, ShowActivated = false, WindowStartupLocation = WindowStartupLocation.Manual };
                    var constructorMs = watch.Elapsed.TotalMilliseconds; var constructorBytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    dialog.Show(); var showMs = watch.Elapsed.TotalMilliseconds - constructorMs; var showBytes = GC.GetAllocatedBytesForCurrentThread() - bytes - constructorBytes;
                    await Idle(dialog); var displayedMs = watch.Elapsed.TotalMilliseconds; var visible = Visible(dialog);
                    var displayedBytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    await Drain(dialog); watch.Stop(); var rawBytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    if (i >= 2) samples.Add(new { constructorMs, constructorBytes, showMs, showBytes, layoutAndQueueMs = displayedMs - constructorMs - showMs, displayedMs, displayedBytes, rawMs = watch.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = watch.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes, idleBytes, correctedBytes = rawBytes - idleBytes, visibleControls = visible });
                    dialog.Close(); await Idle(sentinel);
                }
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "SettingsWindow constructor/Show/ApplicationIdle, 760x700 offscreen WPF; two warmups and seven samples; paired 250ms and ApplicationIdle", samples }, new JsonSerializerOptions { WriteIndented = true }));
                exit = 0;
            }
            catch (Exception ex) { Console.Error.WriteLine(ex); }
            finally { sentinel.Close(); app.Shutdown(exit); }
        };
        sentinel.Show(); app.Run(); return exit;
    }
}
