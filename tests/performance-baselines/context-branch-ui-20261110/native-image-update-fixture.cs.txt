using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

class PerfImageUpdate
{
    [STAThread] static int Main(string[] args)
    {
        var output=Path.GetFullPath(args[0]);Directory.CreateDirectory(output);
        var app=new Application{ShutdownMode=ShutdownMode.OnExplicitShutdown};
        var window=new Window{Width=900,Height=760,Left=20,Top=20,Content=new Grid()};int exit=1;
        window.Loaded+=async(_,_)=>{try{
            var type=typeof(AiPanel).Assembly.GetType("TeXeed.App.Controls.AiAnnotationImages")!;
            var add=type.GetMethod("Add",BindingFlags.Static|BindingFlags.NonPublic)!;
            var create=type.GetMethod("Create",BindingFlags.Static|BindingFlags.NonPublic)!;
            var bytes=new byte[512*512*4];for(int i=0;i<bytes.Length;i+=4){int n=i/4;bytes[i]=(byte)(n%256);bytes[i+1]=(byte)((n/512)%256);bytes[i+2]=(byte)((n^(n/512))%256);bytes[i+3]=255;}
            var bitmap=BitmapSource.Create(512,512,96,96,PixelFormats.Bgra32,null,bytes,512*4);bitmap.Freeze();
            var entry=new AiEntry();for(int i=0;i<5;i++){object?[] arg=[entry,bitmap,null];if(add.Invoke(null,arg) is not true)throw new Exception((string?)arg[2]);}
            var panel=(FrameworkElement)create.Invoke(null,[entry,(Action)(()=>{})])!;window.Content=panel;
            async Task Idle(){await window.Dispatcher.InvokeAsync(()=>{},DispatcherPriority.ApplicationIdle);window.UpdateLayout();}
            int Controls(DependencyObject root){int count=0;for(int i=0;i<VisualTreeHelper.GetChildrenCount(root);i++){var child=VisualTreeHelper.GetChild(root,i);if(child is Control c&&c.IsVisible)count++;count+=Controls(child);}return count;}
            await Idle();var samples=new List<object>();
            for(int i=0;i<9;i++){
                var before=GC.GetAllocatedBytesForCurrentThread();var timer=Stopwatch.StartNew();await Idle();timer.Stop();var idleMs=timer.Elapsed.TotalMilliseconds;var idleBytes=GC.GetAllocatedBytesForCurrentThread()-before;
                before=GC.GetAllocatedBytesForCurrentThread();timer.Restart();object?[] arg=[entry,bitmap,null];if(add.Invoke(null,arg) is not true)throw new Exception((string?)arg[2]);await Idle();timer.Stop();var raw=GC.GetAllocatedBytesForCurrentThread()-before;
                if(i>=2)samples.Add(new{rawMs=timer.Elapsed.TotalMilliseconds,idleMs,correctedMs=timer.Elapsed.TotalMilliseconds-idleMs,rawBytes=raw,idleBytes,correctedBytes=raw-idleBytes,visibleControls=Controls(window)});
                entry.Images.RemoveAt(entry.Images.Count-1);await Idle();
            }
            File.WriteAllText(Path.Combine(output,"results.json"),JsonSerializer.Serialize(new{fixture="existing 5 synthetic 512x512 BGRA images; add sixth; 2 warmups, 7 paired ApplicationIdle samples; WPF 900x760",samples},new JsonSerializerOptions{WriteIndented=true}));exit=0;
        }catch(Exception e){Console.Error.WriteLine(e);}finally{window.Close();app.Shutdown(exit);}};
        window.Show();app.Run();return exit;
    }
}
