class Perf {
const BindingFlags P=BindingFlags.Instance|BindingFlags.NonPublic|BindingFlags.Public;
static object Field(object x,string name)=>x.GetType().GetField(name,P)!.GetValue(x)!;
static object? Invoke(object x,string name,params object?[] a)=>x.GetType().GetMethod(name,P)!.Invoke(x,a);
static async Task Idle(Window w){await w.Dispatcher.InvokeAsync(()=>{},DispatcherPriority.ApplicationIdle);w.UpdateLayout();}
static IEnumerable<T> Children<T>(DependencyObject d) where T:DependencyObject {for(int i=0;i<VisualTreeHelper.GetChildrenCount(d);i++){var c=VisualTreeHelper.GetChild(d,i);if(c is T t)yield return t;foreach(var x in Children<T>(c))yield return x;}}
static IEnumerable<Paragraph> Paragraphs(BlockCollection blocks)=>blocks.SelectMany(b=>b switch{Paragraph p=>[p],Section s=>Paragraphs(s.Blocks),_=>[]});
static int Visible(DependencyObject root,Window w){int count=0;foreach(var c in Children<Control>(root)){if(!c.IsVisible||c.ActualWidth<=0||c.ActualHeight<=0)continue;try{if(c.TransformToAncestor(w).TransformBounds(new Rect(0,0,c.ActualWidth,c.ActualHeight)).IntersectsWith(new Rect(0,0,w.ActualWidth,w.ActualHeight)))count++;}catch(InvalidOperationException){}}return count;}
[STAThread] static int Main(string[] args){
var output=Path.GetFullPath(args[0]);Directory.CreateDirectory(output);Environment.SetEnvironmentVariable("TEXEED_HOME",Path.Combine(output,"profile"));
var app=new Application{ShutdownMode=ShutdownMode.OnExplicitShutdown};app.Resources.MergedDictionaries.Add(new ResourceDictionary{Source=new Uri("/TeXeed;component/Theme.xaml",UriKind.Relative)});
var panel=new AiPanel();typeof(AiPanel).GetMethod("InitializeBranchUi",P)?.Invoke(panel,null);var host=new Window{Width=900,Height=760,Left=20,Top=20,Content=panel};int result=1;
app.DispatcherUnhandledException+=(_,e)=>{Console.Error.WriteLine(e.Exception);e.Handled=true;app.Shutdown(1);};
host.Loaded+=async(_,_)=>{try{
await panel.SetProjectAsync(Path.Combine(output,"fixture.tex"));var book=(AiNotebook)Field(panel,"_book");
var body="# 长回答\n\n"+string.Join("\n\n",Enumerable.Range(0,60).Select(i=>$"第 {i} 段需要理解的定义。"+new string('文',180)));
book.Turns.Add(new AiTurn("解释定义",body));Invoke(panel,"RefreshTranscript");await Idle(host);
var view=Children<MarkdownView>(panel).First(v=>v.Markdown==body);await (Task)view.GetType().GetProperty("RenderTask",P)!.GetValue(view)!;await Idle(host);
var paragraph=Paragraphs(view.Document.Blocks).First(p=>new TextRange(p.ContentStart,p.ContentEnd).Text.Contains("需要理解"));
var handler=new EventHandler<string>((_,_)=>{});var ev=panel.GetType().GetEvent("ConversationOpenRequested",P)!;ev.GetAddMethod(true)!.Invoke(panel,[handler]);
void Select(){view.Selection.Select(paragraph.ContentStart,paragraph.ContentEnd);}
var suppressLinks = args.Length > 1 && args[1] == "suppress";
if(suppressLinks) ((System.Collections.IDictionary)Field(panel,"_answerLayers")).Clear();
var samples=new List<object>();
for(int i=0;i<9;i++){
Select();var eb=GC.GetAllocatedBytesForCurrentThread();var ew=Stopwatch.StartNew();await Idle(host);ew.Stop();eb=GC.GetAllocatedBytesForCurrentThread()-eb;
var bytes=GC.GetAllocatedBytesForCurrentThread();var watch=Stopwatch.StartNew();await (Task)Invoke(panel,"BranchFromSelectionAsync",view)!;await Idle(host);watch.Stop();bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
if(i>=2)samples.Add(new{op="add-branch-existing-parent",rawMs=watch.Elapsed.TotalMilliseconds,idleMs=ew.Elapsed.TotalMilliseconds,correctedMs=watch.Elapsed.TotalMilliseconds-ew.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=eb,correctedBytes=bytes-eb,visibleControls=Visible(host,host)});
}
var heading=Paragraphs(view.Document.Blocks).First(p=>p.Inlines.OfType<Hyperlink>().Any()).Inlines.OfType<Hyperlink>().First();
for(int i=0;i<9;i++){
var eb=GC.GetAllocatedBytesForCurrentThread();var ew=Stopwatch.StartNew();await Idle(host);ew.Stop();eb=GC.GetAllocatedBytesForCurrentThread()-eb;
var bytes=GC.GetAllocatedBytesForCurrentThread();var watch=Stopwatch.StartNew();heading.RaiseEvent(new RoutedEventArgs(Hyperlink.ClickEvent));await Idle(host);heading.RaiseEvent(new RoutedEventArgs(Hyperlink.ClickEvent));await Idle(host);watch.Stop();bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
if(i>=2)samples.Add(new{op="fold-unfold-long-answer",rawMs=watch.Elapsed.TotalMilliseconds,idleMs=ew.Elapsed.TotalMilliseconds,correctedMs=watch.Elapsed.TotalMilliseconds-ew.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=eb,correctedBytes=bytes-eb,visibleControls=Visible(host,host)});
}
File.WriteAllText(Path.Combine(output,"folding.json"),JsonSerializer.Serialize(new{fixture="900x760; 60 paragraphs x 180 chars; 2 warmups + 7 paired ApplicationIdle; branch creation persists to local isolated file",samples},new JsonSerializerOptions{WriteIndented=true}));result=0;
}catch(Exception ex){Console.Error.WriteLine(ex);}finally{panel.Dispose();host.Content=null;host.Close();app.Shutdown(result);}};host.Show();app.Run();return result;}
}
