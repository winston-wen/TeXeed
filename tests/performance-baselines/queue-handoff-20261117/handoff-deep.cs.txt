using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

internal static class HandoffDeep
{
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private static object Field(object x, string name) => x.GetType().GetField(name, Private)!.GetValue(x)!;
    private static void Call(object x, string name) => x.GetType().GetMethod(name, Private)!.Invoke(x, null);
    private static async Task Idle(Window w) { await w.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); w.UpdateLayout(); }
    private static async Task Drain(AiPanel panel, Window w)
    {
        await Task.Delay(800); var timer = (DispatcherTimer)Field(panel, "_saveTimer");
        for (var i = 0; timer.IsEnabled && i < 250; i++) await Task.Delay(20);
        if (timer.IsEnabled) throw new TimeoutException("Draft save did not drain.");
        var gate = (SemaphoreSlim)Field(panel, "_saveGate"); await gate.WaitAsync(); gate.Release(); await Idle(w);
    }
    private static int Visible(DependencyObject root, Window w)
    {
        var count = 0;
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        {
            var child = VisualTreeHelper.GetChild(root, i);
            if (child is Control c && c.IsVisible && c.ActualWidth > 0 && c.ActualHeight > 0)
                try { if (c.TransformToAncestor(w).TransformBounds(new Rect(0, 0, c.ActualWidth, c.ActualHeight)).IntersectsWith(new Rect(0, 0, w.ActualWidth, w.ActualHeight))) count++; } catch (InvalidOperationException) { }
            count += Visible(child, w);
        }
        return count;
    }
    [STAThread] private static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
        Environment.SetEnvironmentVariable("TEXEED_HOME", Path.Combine(output, "profile"));
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        using var panel = new AiPanel(); var window = new Window { Width = 900, Height = 760, Left = -12000, Top = 20, ShowActivated = false, WindowStartupLocation = WindowStartupLocation.Manual, Content = panel };
        var exit = 1;
        app.DispatcherUnhandledException += (_, e) => { Console.Error.WriteLine(e.Exception); e.Handled = true; app.Shutdown(1); };
        window.Loaded += async (_, _) =>
        {
            try
            {
                await panel.SetProjectAsync(Path.Combine(output, "fixture.tex")); var book = (AiNotebook)Field(panel, "_book");
                for (var n = 0; n < 120; n++) book.Turns.Add(new AiTurn("问题 " + n, "样例回答") { SentAt = DateTimeOffset.UnixEpoch });
                Call(panel, "RefreshTranscript"); await Drain(panel, window);
                var samples = new List<object>();
                foreach (var op in new[] { "local-complete", "persisted-complete", "save-only" })
                for (var i = 0; i < 9; i++)
                {
                    var queue = AiFollowUpQueue.Load(book.Active); var first = queue.Enqueue("queue", "queued active", []); var second = queue.Enqueue("queue", "queued waiting", []);
                    if (!queue.MarkDispatching([first.Id], Guid.NewGuid().ToString("N"))) throw new Exception("setup failed");
                    queue.Store(book.Active); Call(panel, "RefreshFollowUps"); await Drain(panel, window);
                    var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(panel, window); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                    var bytes = GC.GetAllocatedBytesForCurrentThread(); var sw = Stopwatch.StartNew();
                    if (op == "local-complete") { var local = AiFollowUpQueue.Load(book.Active); local.Complete([first.Id]); local.Store(book.Active); Call(panel, "RefreshFollowUps"); }
                    else if (op == "persisted-complete")
                    {
                        Func<AiFollowUpQueue, bool> mutation = q => q.Complete([first.Id]);
                        var task = (Task<bool>)typeof(AiPanel).GetMethod("MutateFollowUpsAsync", Private)!.Invoke(panel, [book, mutation])!;
                        if (!await task) throw new Exception("completion failed");
                    }
                    else await panel.SaveAsync();
                    await Idle(window); var feedback = sw.Elapsed.TotalMilliseconds; var immediate = GC.GetAllocatedBytesForCurrentThread() - bytes; var controls = Visible(window, window);
                    await Drain(panel, window); sw.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    if (i >= 2) samples.Add(new { op, feedbackMs = feedback, feedbackBytes = immediate, rawMs = sw.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = sw.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = controls });
                    var cleanup = AiFollowUpQueue.Load(book.Active); cleanup.Complete([first.Id]); cleanup.Remove(second.Id); cleanup.Store(book.Active); Call(panel, "RefreshFollowUps");
                }
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "120 turns, synthetic two-item queue; compare in-memory completion + redraw, persisted completion + redraw, and standalone save; 900x760 offscreen; two warmups and seven samples; paired 800ms and save gate", samples }, new JsonSerializerOptions { WriteIndented = true }));
                exit = 0;
            }
            catch (Exception ex) { Console.Error.WriteLine(ex); }
            finally { window.Close(); app.Shutdown(exit); }
        };
        window.Show(); app.Run(); return exit;
    }
}
