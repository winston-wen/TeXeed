using System.Diagnostics;
using System.IO;
using System.Net;
using System.Net.Http;
using System.Reflection;
using System.Text;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

internal static class QueuePerf
{
    private const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    private static object Field(object instance, string name) => instance.GetType().GetField(name, Private)!.GetValue(instance)!;
    private static async Task CallAsync(object instance, string name, params object[] args) => await (Task)instance.GetType().GetMethod(name, Private)!.Invoke(instance, args)!;
    private sealed class Handler : HttpMessageHandler
    {
        internal readonly TaskCompletionSource FirstEntered = new(TaskCreationOptions.RunContinuationsAsynchronously);
        internal readonly TaskCompletionSource ReleaseFirst = new(TaskCreationOptions.RunContinuationsAsynchronously);
        internal readonly TaskCompletionSource SecondEntered = new(TaskCreationOptions.RunContinuationsAsynchronously);
        internal int Count;
        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken token)
        {
            var n = Interlocked.Increment(ref Count);
            if (n == 1) { FirstEntered.TrySetResult(); await ReleaseFirst.Task.WaitAsync(token); }
            if (n == 2) { SecondEntered.TrySetResult(); await Task.Delay(Timeout.Infinite, token); }
            return new(HttpStatusCode.OK) { Content = new StringContent("data: {\"type\":\"response.output_text.delta\",\"delta\":\"done\"}\n\ndata: {\"type\":\"response.completed\",\"response\":{\"output\":[]}}\n\n", Encoding.UTF8, "text/event-stream") };
        }
    }
    private static async Task Idle(Window window)
    { await window.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); window.UpdateLayout(); }
    private static async Task Drain(AiPanel panel, Window window)
    {
        await Task.Delay(800);
        var timer = (DispatcherTimer)Field(panel, "_saveTimer");
        for (var n = 0; timer.IsEnabled && n < 250; n++) await Task.Delay(20);
        if (timer.IsEnabled) throw new TimeoutException("Draft save did not drain.");
        var gate = (SemaphoreSlim)Field(panel, "_saveGate"); await gate.WaitAsync(); gate.Release(); await Idle(window);
    }
    private static int Visible(DependencyObject root, Window window)
    {
        var result = 0;
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        {
            var child = VisualTreeHelper.GetChild(root, i);
            if (child is Control c && c.IsVisible && c.ActualWidth > 0 && c.ActualHeight > 0)
                try { if (c.TransformToAncestor(window).TransformBounds(new Rect(0, 0, c.ActualWidth, c.ActualHeight)).IntersectsWith(new Rect(0, 0, window.ActualWidth, window.ActualHeight))) result++; } catch (InvalidOperationException) { }
            result += Visible(child, window);
        }
        return result;
    }
    [STAThread] private static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
        Environment.SetEnvironmentVariable("TEXEED_HOME", Path.Combine(output, "profile"));
        new AiCredentials { ApiKey = "fixture-not-real", UseApiKey = true }.Save();
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        var sentinel = new Window { Width = 1, Height = 1, Left = -12000, Top = 20, ShowActivated = false, WindowStartupLocation = WindowStartupLocation.Manual };
        var exit = 1;
        app.DispatcherUnhandledException += (_, e) => { Console.Error.WriteLine(e.Exception); e.Handled = true; app.Shutdown(1); };
        sentinel.Loaded += async (_, _) =>
        {
            try
            {
                var samples = new List<object>();
                for (var i = 0; i < 9; i++)
                {
                    using var handler = new Handler(); using var panel = new AiPanel(new AiConnection(handler));
                    await panel.SetProjectAsync(Path.Combine(output, "project-" + i + ".tex"));
                    var models = (ComboBox)panel.Settings.FindName("Models"); models.ItemsSource = new[] { new AiModel("fixture", "测试模型") }; models.SelectedIndex = 0;
                    var book = (AiNotebook)Field(panel, "_book");
                    for (var n = 0; n < 120; n++) book.Turns.Add(new AiTurn("问题 " + n, "样例回答") { SentAt = DateTimeOffset.UnixEpoch });
                    var window = new Window { Width = 900, Height = 760, Left = -12000, Top = 20, ShowActivated = false, WindowStartupLocation = WindowStartupLocation.Manual, Content = panel };
                    window.Show(); await Idle(window);
                    var composer = (TextBox)panel.FindName("Composer"); composer.Text = "first question";
                    var first = panel.SubmitAsync(); await handler.FirstEntered.Task.WaitAsync(TimeSpan.FromSeconds(30));
                    composer.Text = "queued active"; await CallAsync(panel, "AddFollowUpAsync", "queue");
                    composer.Text = "queued waiting"; await CallAsync(panel, "AddFollowUpAsync", "queue");
                    composer.Text = "unsent draft"; await Drain(panel, window);
                    var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(panel, window); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                    var bytes = GC.GetAllocatedBytesForCurrentThread(); var sw = Stopwatch.StartNew();
                    handler.ReleaseFirst.TrySetResult(); await handler.SecondEntered.Task.WaitAsync(TimeSpan.FromSeconds(30)); await Idle(window);
                    var feedback = sw.Elapsed.TotalMilliseconds; var immediate = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    var dock = (FrameworkElement)panel.FindName("FollowUpDock"); var rows = (Panel)panel.FindName("FollowUpItems");
                    var labels = rows.Children.OfType<DockPanel>().SelectMany(row => row.Children.OfType<TextBlock>()).Select(text => text.Text).ToArray();
                    var live = (FrameworkElement)panel.FindName("LiveContainer");
                    var visible = Visible(window, window);
                    await Drain(panel, window); sw.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    if (!live.IsVisible || handler.Count != 2 || composer.Text != "unsent draft") throw new Exception("Handoff did not create the expected active turn or preserved draft.");
                    if (i >= 2) samples.Add(new { feedbackMs = feedback, feedbackBytes = immediate, rawMs = sw.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = sw.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = visible, dockVisible = dock.IsVisible, rowCount = rows.Children.Count, labels });
                    await panel.StopAndSaveAsync(); await first; window.Close();
                }
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "120 turns; queue two messages while first response is held; release first, wait for second request and measure queue UI; 900x760 offscreen; two warmups and seven samples; paired 800ms queue and save gate", samples }, new JsonSerializerOptions { WriteIndented = true }));
                exit = 0;
            }
            catch (Exception ex) { Console.Error.WriteLine(ex); }
            finally { sentinel.Close(); app.Shutdown(exit); }
        };
        sentinel.Show(); app.Run(); return exit;
    }
}
