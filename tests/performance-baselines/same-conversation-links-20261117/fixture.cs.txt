using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

class SameSessionLinkPerf
{
    const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.Public;
    static object Field(object value, string name) => value.GetType().GetField(name, Private)!.GetValue(value)!;
    static object? Call(object value, string name, params object?[] args) => value.GetType().GetMethod(name, Private)!.Invoke(value, args);
    static async Task Idle(Window window) { await window.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); window.UpdateLayout(); }
    static async Task Drain(Window window) { await Task.Delay(800); await Idle(window); }
    static IEnumerable<T> Children<T>(DependencyObject root) where T : DependencyObject
    {
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(root); i++)
        {
            var child = VisualTreeHelper.GetChild(root, i);
            if (child is T match) yield return match;
            foreach (var nested in Children<T>(child)) yield return nested;
        }
    }
    static int Visible(Window window) => Children<Control>(window).Count(control =>
    {
        if (!control.IsVisible || control.ActualWidth <= 0 || control.ActualHeight <= 0) return false;
        try { return control.TransformToAncestor(window).TransformBounds(new Rect(0, 0, control.ActualWidth, control.ActualHeight)).IntersectsWith(new Rect(0, 0, window.ActualWidth, window.ActualHeight)); }
        catch (InvalidOperationException) { return false; }
    });

    [STAThread] static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
        var mode = args.Length > 1 ? args[1] : "";
        var expectLinks = mode == "links";
        var staleAnchors = mode == "invalid";
        var startupOnly = args.Length > 2 && args[2] == "startup-only";
        Environment.SetEnvironmentVariable("TEXEED_HOME", Path.Combine(output, "profile"));
        var app = new Application { ShutdownMode = ShutdownMode.OnExplicitShutdown };
        app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source = new Uri("/TeXeed;component/Theme.xaml", UriKind.Relative) });
        var panel = new AiPanel(); typeof(AiPanel).GetMethod("InitializeBranchUi", Private)?.Invoke(panel, null);
        var host = new Window { Width = 900, Height = 760, Left = 20, Top = 20, Content = panel }; var exit = 1;
        app.DispatcherUnhandledException += (_, e) => { Console.Error.WriteLine(e.Exception); e.Handled = true; app.Shutdown(1); };
        host.Loaded += async (_, _) =>
        {
            try
            {
                await panel.SetProjectAsync(Path.Combine(output, "fixture.tex"));
                var book = (AiNotebook)Field(panel, "_book");
                var fragments = Enumerable.Range(1, 8).Select(i => $"来源选区{i}号").ToArray();
                var body = string.Join("，", fragments) + "。这是用于比较虚拟列表恢复的合成回答。";
                var segment = new AiSegment { Id = "source-segment", Kind = "final", Text = body };
                var source = new AiTurn("原问题", body) { Id = "source-turn", SentAt = DateTimeOffset.UnixEpoch, Segments = [segment] };
                book.Turns.Add(source);
                var revision = Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(body)));
                for (var i = 1; i < 120; i++)
                {
                    var entries = new List<AiEntry>();
                    if (i <= fragments.Length)
                    {
                        var quote = fragments[i - 1]; var offset = body.IndexOf(quote, StringComparison.Ordinal);
                        entries.Add(new AiEntry { AnswerAnchor = new AiAnswerAnchor(source.Id, segment.Id, offset, quote.Length, quote, staleAnchors ? "stale-revision" : revision) { TextOffset = offset } });
                    }
                    book.Turns.Add(new AiTurn("继续解释 " + i, "合成回答 " + i) { Id = "turn-" + i, SentAt = DateTimeOffset.UnixEpoch, SentAnnotations = entries });
                }
                Call(panel, "RefreshTranscript"); await Drain(host);
                var scroll = (ScrollViewer)panel.FindName("MessagesScroll");
                var samples = new List<object>();
                async Task ToSource()
                {
                    scroll.ScrollToVerticalOffset(0); await Drain(host);
                    if (!Children<AiTurnView>(panel).Any(view => view.Turn.Id == source.Id)) throw new Exception("Source turn was not realized");
                }
                async Task MeasureRestore(string op)
                {
                    for (var i = 0; i < 9; i++)
                    {
                        scroll.ScrollToEnd(); await Drain(host);
                        var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(host); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                        var bytes = GC.GetAllocatedBytesForCurrentThread(); var watch = Stopwatch.StartNew(); scroll.ScrollToVerticalOffset(0); await Idle(host);
                        var feedback = watch.Elapsed.TotalMilliseconds; await Drain(host); watch.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                        var visible = Visible(host);
                        if (!host.IsVisible || host.WindowState == WindowState.Minimized || visible == 0) throw new Exception($"Host is not visible: IsVisible={host.IsVisible}, state={host.WindowState}, controls={visible}, size={host.ActualWidth}x{host.ActualHeight}");
                        var sourceView = Children<AiTurnView>(panel).FirstOrDefault(view => view.Turn.Id == source.Id) ?? throw new Exception("Scroll did not restore source");
                        var links = Children<Button>(sourceView).Count(button => (button.Content?.ToString() ?? "").StartsWith("🗨️ [", StringComparison.Ordinal));
                        if (expectLinks && links != 8) throw new Exception($"Expected eight source links; found {links}");
                        if (i >= 2) samples.Add(new { op, feedbackMs = feedback, rawMs = watch.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = watch.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = Visible(host), links });
                    }
                }
                if (!startupOnly) await MeasureRestore("restore-source-with-references");
                if (expectLinks && !startupOnly)
                {
                    for (var i = 0; i < 9; i++)
                    {
                        await ToSource();
                        var sourceView = Children<AiTurnView>(panel).First(view => view.Turn.Id == source.Id);
                        var link = Children<Button>(sourceView).First(button => (button.Content?.ToString() ?? "") == "🗨️ [2]");
                        var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(host); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                        var bytes = GC.GetAllocatedBytesForCurrentThread(); var watch = Stopwatch.StartNew(); link.RaiseEvent(new RoutedEventArgs(Button.ClickEvent)); await Idle(host);
                        var feedback = watch.Elapsed.TotalMilliseconds; await Drain(host); watch.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                        var target = Children<AiTurnView>(panel).FirstOrDefault(view => view.Turn.Id == "turn-1") ?? throw new Exception("Link did not realize target turn");
                        var top = target.TranslatePoint(new Point(), scroll).Y;
                        if (top >= scroll.ViewportHeight || top + target.ActualHeight <= 0) throw new Exception("Link target is outside the viewport");
                        if (i >= 2) samples.Add(new { op = "open-same-session-reference", feedbackMs = feedback, rawMs = watch.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = watch.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = Visible(host), targetTop = top });
                    }
                }
                var turns = book.Turns.ToArray(); host.Content = null; await Idle(host);
                for (var i = 0; i < 9; i++)
                {
                    using var fresh = new AiPanel(); typeof(AiPanel).GetMethod("InitializeBranchUi", Private)?.Invoke(fresh, null);
                    host.Content = fresh;
                    await fresh.SetProjectAsync(Path.Combine(output, "fresh-" + i + ".tex"));
                    var freshBook = (AiNotebook)Field(fresh, "_book"); freshBook.Turns.AddRange(turns);
                    await Drain(host);
                    var idleBytes = GC.GetAllocatedBytesForCurrentThread(); var idle = Stopwatch.StartNew(); await Drain(host); idle.Stop(); idleBytes = GC.GetAllocatedBytesForCurrentThread() - idleBytes;
                    var bytes = GC.GetAllocatedBytesForCurrentThread(); var watch = Stopwatch.StartNew(); Call(fresh, "RefreshTranscript"); await Idle(host);
                    var feedback = watch.Elapsed.TotalMilliseconds; await Drain(host); watch.Stop(); bytes = GC.GetAllocatedBytesForCurrentThread() - bytes;
                    var freshSource = Children<AiTurnView>(fresh).FirstOrDefault(view => view.Turn.Id == source.Id) ?? throw new Exception("Initial source turn was not realized");
                    var links = Children<Button>(freshSource).Count(button => (button.Content?.ToString() ?? "").StartsWith("🗨️ [", StringComparison.Ordinal));
                    if (expectLinks && links != 8) throw new Exception($"Expected eight initially constructed links; found {links}");
                    if (staleAnchors && links != 0) throw new Exception($"Stale-anchor counterfactual unexpectedly constructed {links} links");
                    if (i >= 2) samples.Add(new { op = "initial-restore-with-references", feedbackMs = feedback, rawMs = watch.Elapsed.TotalMilliseconds, idleMs = idle.Elapsed.TotalMilliseconds, correctedMs = watch.Elapsed.TotalMilliseconds - idle.Elapsed.TotalMilliseconds, rawBytes = bytes, idleBytes, correctedBytes = bytes - idleBytes, visibleControls = Visible(host), links });
                    host.Content = null; await Idle(host);
                }
                File.WriteAllText(Path.Combine(output, "results.json"), JsonSerializer.Serialize(new { fixture = "120 turns; eight references to one source; 900x760; two warmups and seven paired 800ms samples", samples }, new JsonSerializerOptions { WriteIndented = true }));
                exit = 0;
            }
            catch (Exception error) { Console.Error.WriteLine(error); }
            finally { panel.Dispose(); host.Content = null; host.Close(); app.Shutdown(exit); }
        };
        host.Show(); app.Run(); return exit;
    }
}
