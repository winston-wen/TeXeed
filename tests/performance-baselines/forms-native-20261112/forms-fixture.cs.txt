using System.Diagnostics;
using System.IO;
using System.Reflection;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Controls.Primitives;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.AI;
using TeXeed.App.Controls;

class Perf
{
    const BindingFlags Private = BindingFlags.Instance | BindingFlags.NonPublic;
    static object Field(object x,string name)=>x.GetType().GetField(name,Private)!.GetValue(x)!;
    static void Call(object x,string name,params object?[] args)=>x.GetType().GetMethod(name,Private)!.Invoke(x,args);
    static async Task Idle(Window w) { await w.Dispatcher.InvokeAsync(()=>{}, DispatcherPriority.ApplicationIdle); w.UpdateLayout(); }
    static IEnumerable<T> Descendants<T>(DependencyObject root) where T : DependencyObject { for (int i=0; i<VisualTreeHelper.GetChildrenCount(root); i++) { var child=VisualTreeHelper.GetChild(root,i); if (child is T match) yield return match; foreach (var nested in Descendants<T>(child)) yield return nested; } }
    static async Task Drain(AiPanel panel, Window w) { await Task.Delay(800); var timer=(DispatcherTimer)Field(panel,"_saveTimer"); for(int i=0;timer.IsEnabled && i<250;i++) await Task.Delay(20); if(timer.IsEnabled) throw new TimeoutException("The draft save timer did not drain."); var gate=(SemaphoreSlim)Field(panel,"_saveGate"); await gate.WaitAsync(); gate.Release(); await Idle(w); }
    static int Visible(DependencyObject root,Window host) { int count=0; for(int i=0;i<VisualTreeHelper.GetChildrenCount(root);i++) { var child=VisualTreeHelper.GetChild(root,i); if(child is Control c && c.IsVisible && c.ActualWidth>0 && c.ActualHeight>0) { try { var rect=c.TransformToAncestor(host).TransformBounds(new Rect(0,0,c.ActualWidth,c.ActualHeight)); if(rect.IntersectsWith(new Rect(0,0,host.ActualWidth,host.ActualHeight))) count++; } catch(InvalidOperationException){} } count+=Visible(child,host); } return count; }
    [STAThread] static int Main(string[] args)
    {
        var output=Path.GetFullPath(args[0]); Directory.CreateDirectory(output);
        Environment.SetEnvironmentVariable("TEXEED_HOME",Path.Combine(output,"profile"));
        var app=new Application{ShutdownMode=ShutdownMode.OnExplicitShutdown};
        app.Resources.MergedDictionaries.Add(new ResourceDictionary{Source=new Uri("/TeXeed;component/Theme.xaml",UriKind.Relative)});
        var panel=new AiPanel(); typeof(AiPanel).GetMethod("InitializeBranchUi", Private)?.Invoke(panel, null); var w=new Window{Width=900,Height=760,Left=20,Top=20,WindowStartupLocation=WindowStartupLocation.Manual,Content=panel}; int exit=1;
        app.DispatcherUnhandledException+=(_,e)=>{Console.Error.WriteLine(e.Exception);e.Handled=true;app.Shutdown(1);};
        w.Loaded+=async(_,_)=>{try{
            await panel.SetProjectAsync(Path.Combine(output,"fixture.tex")); var book=(AiNotebook)Field(panel,"_book");
            var exercise="| 题号 | 输入 | a | b | c |\n|---|---|---|---|---|\n" + string.Join("\n", Enumerable.Range(1,6).Select(i=>$"| {i} | {80+i} | {{{{text:a-{i}}}}} | {{{{text:b-{i}}}}} | {{{{text:c-{i}}}}} |")) + "\n\n说明：\n\n{{textarea:reason}}\n\n{{text:solo}}\n\n{{radio:choice:yes:是}} {{radio:choice:no:否}} {{checkbox:check:核对}}\n\n{{file:materials:请提供材料}}";
            for(int i=0;i<120;i++) { var text=i==119?exercise:"样例回答：说明步骤和依据。"; var turn=new AiTurn("如何检查结果？ "+i,text) { SentAt=DateTimeOffset.UnixEpoch,Segments=Enumerable.Range(0,3).Select(k=>new AiSegment{Kind="activity",Text="检查步骤 "+k,Details=string.Join('\n',Enumerable.Range(0,200).Select(n=>"示例日志 "+n))}).Append(new AiSegment{Kind="final",Text=text}).ToList() }; book.Turns.Add(AiForms.WithSubmission(turn,null)); }
            Call(panel,"RefreshTranscript"); await Drain(panel,w);
            var scroll=(ScrollViewer)panel.FindName("MessagesScroll"); var composer=(TextBox)panel.FindName("Composer"); var thumb=(Thumb)panel.FindName("ComposerResize");
            scroll.ScrollToEnd(); await Drain(panel,w);
            var input=Descendants<TextBox>(panel).FirstOrDefault(t=>System.Windows.Automation.AutomationProperties.GetName(t)=="a-1") ?? throw new Exception("Form field a-1 was not rendered.");
            var numbers=Descendants<TextBoxLineNumbers>(panel).FirstOrDefault(n=>n.MinimumLineCount==3) ?? throw new Exception("Three-line form gutter was not rendered.");
            if (!numbers.PaintedNumbers.SequenceEqual(new[]{1,2,3})) throw new Exception("Three-line form gutter does not display 1, 2, 3.");
            var samples=new List<object>();
            void Checkpoint() => File.WriteAllText(Path.Combine(output,"results.json"),JsonSerializer.Serialize(new{fixture="120 turns, 3 activities, 200 detail lines; 900/620 x760 WPF; two warmups and seven samples; paired 800ms queue and save gate",samples},new JsonSerializerOptions{WriteIndented=true}));
            async Task Measure(string op,Action<int> act){for(int i=0;i<9;i++){
                var qb=GC.GetAllocatedBytesForCurrentThread();var qw=Stopwatch.StartNew();await Drain(panel,w);qw.Stop();qb=GC.GetAllocatedBytesForCurrentThread()-qb;
                var bytes=GC.GetAllocatedBytesForCurrentThread();var sw=Stopwatch.StartNew();act(i);await Idle(w);var feedback=sw.Elapsed.TotalMilliseconds;var immediate=GC.GetAllocatedBytesForCurrentThread()-bytes;await Drain(panel,w);sw.Stop();bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
                if(i>=2){samples.Add(new{op,feedbackMs=feedback,feedbackBytes=immediate,rawMs=sw.Elapsed.TotalMilliseconds,idleMs=qw.Elapsed.TotalMilliseconds,correctedMs=sw.Elapsed.TotalMilliseconds-qw.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=qb,correctedBytes=bytes-qb,visibleControls=Visible(w,w)}); Checkpoint();}
            }}
            await Measure("form-input",i=>input.Text=i%2==0?"83":"84" );
            await Measure("resize",i=>{w.Width=i%2==0?620:900;});
            await Measure("scroll",i=>scroll.ScrollToVerticalOffset(i%2==0?0:scroll.ScrollableHeight*.6));
            await Measure("composer-height",i=>thumb.RaiseEvent(new DragDeltaEventArgs(0,i%2==0?-20:20){RoutedEvent=Thumb.DragDeltaEvent}));
            scroll.ScrollToEnd(); await Drain(panel,w);
            var area=Descendants<TextBox>(panel).FirstOrDefault(t=>System.Windows.Automation.AutomationProperties.GetName(t)=="reason") ?? throw new Exception("Multiline form field reason was not rendered.");
            var longInput=string.Join('\n',Enumerable.Range(1,200).Select(i=>$"第 {i} 行填写文字"));
            area.BringIntoView(); await Drain(panel,w);
            await Measure("multiline-200-lines",i=>area.Text=longInput+(i%2==0?"甲":"乙"));
            area.Text=""; scroll.ScrollToEnd(); await Drain(panel,w);
            if (!Descendants<Button>(panel).Any(t=>System.Windows.Automation.AutomationProperties.GetName(t)=="添加附件：请提供材料")) throw new Exception("Form attachment control was not rendered.");
            var folder=Path.Combine(output,"sample-folder"); Directory.CreateDirectory(folder);
            var last=book.Turns[^1]; var origin=new AiFormAttachmentOrigin(1,book.Active.Id,book.Active.Title,last.Id,book.Turns.Count,"materials","请提供材料");
            for(int i=0;i<9;i++) {
                foreach(var entry in book.Entries.ToArray()) panel.RemoveAnnotation(entry);
                await Drain(panel,w); var qb=GC.GetAllocatedBytesForCurrentThread(); var qw=Stopwatch.StartNew(); await Drain(panel,w); qw.Stop(); qb=GC.GetAllocatedBytesForCurrentThread()-qb;
                var bytes=GC.GetAllocatedBytesForCurrentThread(); var sw=Stopwatch.StartNew(); if(!await panel.AddAttachmentsAsync([folder],origin,()=>true)) throw new Exception("Directory reference failed."); await Idle(w); var feedback=sw.Elapsed.TotalMilliseconds; await Drain(panel,w); sw.Stop(); bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
                if(i>=2) {samples.Add(new{op="attach-directory",feedbackMs=feedback,feedbackBytes=0L,rawMs=sw.Elapsed.TotalMilliseconds,idleMs=qw.Elapsed.TotalMilliseconds,correctedMs=sw.Elapsed.TotalMilliseconds-qw.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=qb,correctedBytes=bytes-qb,visibleControls=Visible(w,w)}); Checkpoint();}
            }
            var secondText="另一份练习：{{text:grade-b}}";
            var second=AiForms.WithSubmission(new AiTurn("另一份练习",secondText) { SentAt=DateTimeOffset.UnixEpoch }, null);
            AiForms.SetDraft(last,"text","a-1","83"); AiForms.SetDraft(second,"text","grade-b","42");
            book.Turns.Add(second); var sourceCount=book.Turns.Count;
            var stateA=last.AdditionalData; var stateB=second.AdditionalData;
            void ResetForms() { while(book.Turns.Count>sourceCount) book.Turns.RemoveAt(book.Turns.Count-1); last.AdditionalData=stateA; second.AdditionalData=stateB; }
            AiForms.Submission[] Snapshots() => [AiForms.Snapshot(last)!,AiForms.Snapshot(second)!];
            AiTurn MakeResponse(AiForms.Submission[] snapshots) {
                var sent=AiForms.WithSubmissions(new AiTurn("检查两份答案","") {SentAt=DateTimeOffset.Now},snapshots);
                var forms=new[]{new{turn=last.Id,submission=sent.Id,fields=new Dictionary<string,object>{["text:a-1"]=new{result="correct",comment="已核对"}}},new{turn=second.Id,submission=sent.Id,fields=new Dictionary<string,object>{["text:grade-b"]=new{result="incorrect",comment="再检查"}}}};
                return sent with {Answer="已核对。\n\n```texeed-grade\n"+JsonSerializer.Serialize(new{forms})+"\n```",AnsweredAt=DateTimeOffset.Now,Incomplete=false};
            }
            async Task MeasureTwoForms(string op, Action setup, Action action) { for(int i=0;i<9;i++) {
                setup(); await Drain(panel,w); var qb=GC.GetAllocatedBytesForCurrentThread(); var qw=Stopwatch.StartNew(); await Drain(panel,w); qw.Stop(); qb=GC.GetAllocatedBytesForCurrentThread()-qb;
                var bytes=GC.GetAllocatedBytesForCurrentThread(); var sw=Stopwatch.StartNew(); action(); await Idle(w); var feedback=sw.Elapsed.TotalMilliseconds; await Drain(panel,w); sw.Stop(); bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
                if(i>=2) {samples.Add(new{op,feedbackMs=feedback,feedbackBytes=0L,rawMs=sw.Elapsed.TotalMilliseconds,idleMs=qw.Elapsed.TotalMilliseconds,correctedMs=sw.Elapsed.TotalMilliseconds-qw.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=qb,correctedBytes=bytes-qb,visibleControls=Visible(w,w)}); Checkpoint();}
            } }
            await MeasureTwoForms("submit-two-forms",ResetForms,()=> {var snapshots=Snapshots(); var sent=AiForms.WithSubmissions(new AiTurn("检查两份答案",""){SentAt=DateTimeOffset.Now},snapshots); book.Turns.Add(sent); foreach(var s in snapshots) AiForms.Lock(book.Active,s); Call(panel,"RefreshFormEnabled");});
            AiTurn response=null!;
            await MeasureTwoForms("grade-two-forms",()=> {ResetForms(); var snapshots=Snapshots(); response=MakeResponse(snapshots); book.Turns.Add(response); foreach(var s in snapshots) AiForms.Lock(book.Active,s); Call(panel,"RefreshTranscript"); scroll.ScrollToEnd();},()=> {if(!AiFormGrading.Apply(book.Active,response)) throw new Exception("Two-form grading failed."); Call(panel,"RefreshFormEnabled");});
            if(!AiFormGrading.IsReopened(book.Active,last.Id)||!AiFormGrading.IsReopened(book.Active,second.Id)) throw new Exception("Graded forms did not reopen.");
            var graded=Descendants<TextBox>(panel).FirstOrDefault(t=>System.Windows.Automation.AutomationProperties.GetName(t)=="grade-b") ?? throw new Exception("Graded field is not rendered.");
            if(graded.IsReadOnly || !System.Windows.Automation.AutomationProperties.GetHelpText(graded).Contains("错误")) throw new Exception("Grading marker or editability is missing.");
            await Measure("graded-edit",i=>graded.Text=i%2==0?"41":"42");
            File.WriteAllText(Path.Combine(output,"results.json"),JsonSerializer.Serialize(new{fixture="120 turns, 3 activities, 200 detail lines; 900/620 x760 WPF; two warmups and seven samples; paired 800ms queue and save gate",samples},new JsonSerializerOptions{WriteIndented=true}));
            exit=0;
        }catch(Exception e){Console.Error.WriteLine(e);}finally{panel.Dispose();w.Close();app.Shutdown(exit);}};
        w.Show();app.Run();return exit;
    }
}
