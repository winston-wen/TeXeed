using System.Diagnostics;
using System.IO;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;
using TeXeed.App.Controls;

class Perf
{
    static async Task Idle(Window w) { await w.Dispatcher.InvokeAsync(() => { }, DispatcherPriority.ApplicationIdle); w.UpdateLayout(); }
    static long Private() { using var p = Process.GetCurrentProcess(); p.Refresh(); return p.PrivateMemorySize64; }
    static int Images(DependencyObject o) { var n = 0; for (int i=0;i<VisualTreeHelper.GetChildrenCount(o);i++) { var c=VisualTreeHelper.GetChild(o,i); if (c is Image image && image.IsVisible) n++; n += Images(c); } return n; }
    [STAThread] static int Main(string[] args)
    {
        var output=Path.GetFullPath(args[0]); Directory.CreateDirectory(output); Environment.SetEnvironmentVariable("TEXEED_HOME",Path.Combine(output,"profile"));
        var app=new Application { ShutdownMode=ShutdownMode.OnExplicitShutdown }; app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source=new Uri("/TeXeed;component/Theme.xaml",UriKind.Relative) });
        var view=new MarkdownView { ChatMode=true }; var w=new Window { Width=900,Height=760,Left=20,Top=20,Content=view }; int exit=1;
        w.Loaded+=async(_,_)=>{ try {
            await Idle(w); var rows=new List<object>();
            var plain=string.Join("\n\n",Enumerable.Range(0,50).Select(i=>$"第 {i} 段是用于排版的普通中文文字。"));
            var samples=new[]{("plain",plain),("one",plain+" 😀"),("many",plain+" "+string.Concat(Enumerable.Repeat("😀 🗨️ 👩🏽‍💻 🇨🇳 ",15)))};
            async Task Measure(string name,string text,int index) { await Idle(w); var idle=Stopwatch.StartNew(); var idleBytes=GC.GetAllocatedBytesForCurrentThread(); await Idle(w); idle.Stop(); idleBytes=GC.GetAllocatedBytesForCurrentThread()-idleBytes; var before=Private(); var bytes=GC.GetAllocatedBytesForCurrentThread(); var sw=Stopwatch.StartNew(); view.Markdown=text; await view.RenderTask; await Idle(w); sw.Stop(); var after=Private(); rows.Add(new {scenario=name,index,elapsedMs=sw.Elapsed.TotalMilliseconds,idleMs=idle.Elapsed.TotalMilliseconds,correctedMs=sw.Elapsed.TotalMilliseconds-idle.Elapsed.TotalMilliseconds,allocated=GC.GetAllocatedBytesForCurrentThread()-bytes,idleBytes,privateBefore=before,privateAfter=after,visibleImages=Images(view)}); }
            for (int i=0;i<samples.Length;i++) await Measure(samples[i].Item1,samples[i].Item2,-1);
            for (int i=0;i<9;i++) foreach(var s in samples) { view.Markdown=""; await Idle(w); await Measure(s.Item1,s.Item2,i); }
            File.WriteAllText(Path.Combine(output,"results.json"),JsonSerializer.Serialize(new { fixture="50 paragraphs; plain, one and sixty emoji; one cold and 2 warmups + 7 samples; adjacent ApplicationIdle",rows },new JsonSerializerOptions{WriteIndented=true})); exit=0;
        } catch(Exception e) { Console.Error.WriteLine(e); } finally { w.Close(); app.Shutdown(exit); } };
        w.Show(); app.Run(); return exit;
    }
}
