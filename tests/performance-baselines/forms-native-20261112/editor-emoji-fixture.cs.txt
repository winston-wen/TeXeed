using System.Diagnostics;
using System.IO;
using System.Text.Json;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Media.Imaging;
using System.Windows.Threading;
using TeXeed.App.Controls;
using TeXeed.Core;

class Perf
{
    static async Task Idle(Window w) { await w.Dispatcher.InvokeAsync(()=>{}, DispatcherPriority.ApplicationIdle); w.UpdateLayout(); }
    static async Task Drain(Window w) { await Task.Delay(80); await Idle(w); }
    static int Visible(DependencyObject root, Window host) { int count=0; for(int i=0;i<VisualTreeHelper.GetChildrenCount(root);i++) { var child=VisualTreeHelper.GetChild(root,i); if(child is Control c && c.IsVisible && c.ActualWidth>0 && c.ActualHeight>0) { try { var r=c.TransformToAncestor(host).TransformBounds(new Rect(0,0,c.ActualWidth,c.ActualHeight)); if(r.IntersectsWith(new Rect(0,0,host.ActualWidth,host.ActualHeight))) count++; } catch(InvalidOperationException){} } count+=Visible(child,host); } return count; }
    [STAThread] static int Main(string[] args)
    {
        var output=Path.GetFullPath(args[0]); Directory.CreateDirectory(output); Environment.SetEnvironmentVariable("TEXEED_HOME",Path.Combine(output,"profile"));
        var app=new Application { ShutdownMode=ShutdownMode.OnExplicitShutdown }; app.Resources.MergedDictionaries.Add(new ResourceDictionary { Source=new Uri("/TeXeed;component/Theme.xaml",UriKind.Relative) });
        var editor=new LatexEditor { FontSize=20 }; var window=new Window { Width=900,Height=760,Left=20,Top=20,Content=editor }; int exit=1;
        window.Loaded+=async(_,_)=>{try {
            var results=new List<object>();
            foreach(var style in new[]{"plain","emoji"}) {
                var content=string.Join('\n', Enumerable.Range(1,1000).Select(i => $"\\textbf{{第 {i} 行：正文}} {(style=="emoji"?"😀 🗨️ 👩🏽‍💻 🇨🇳":"示例说明")}"));
                editor.ShowFile(new ProjectFile(style+".tex",style+".tex",ProjectFileKind.Tex,content),content);
                editor.ScrollToVerticalOffset(0); await Drain(window);
                var view=editor.TextArea.TextView; var width=Math.Max(1,(int)Math.Ceiling(view.ActualWidth)); var height=Math.Max(1,(int)Math.Ceiling(view.ActualHeight));
                async Task Measure(string operation,Action<int> action) { for(int i=0;i<9;i++) {
                    var qb=GC.GetAllocatedBytesForCurrentThread(); var qw=Stopwatch.StartNew(); await Drain(window); qw.Stop(); qb=GC.GetAllocatedBytesForCurrentThread()-qb;
                    var bytes=GC.GetAllocatedBytesForCurrentThread(); var sw=Stopwatch.StartNew(); action(i); await Idle(window); var feedback=sw.Elapsed.TotalMilliseconds; await Drain(window); sw.Stop(); bytes=GC.GetAllocatedBytesForCurrentThread()-bytes;
                    if(i>=2) results.Add(new { style,operation,feedbackMs=feedback,rawMs=sw.Elapsed.TotalMilliseconds,idleMs=qw.Elapsed.TotalMilliseconds,correctedMs=sw.Elapsed.TotalMilliseconds-qw.Elapsed.TotalMilliseconds,rawBytes=bytes,idleBytes=qb,correctedBytes=bytes-qb,visibleControls=Visible(window,window) });
                } }
                await Measure("draw",i=>{ var bitmap=new RenderTargetBitmap(width,height,96,96,PixelFormats.Pbgra32); bitmap.Render(view); });
                editor.ScrollToVerticalOffset(0); await Drain(window);
                var offset=editor.Document.GetLineByNumber(10).Offset;
                await Measure("input",i=>editor.Document.Replace(offset,1,i%2==0?"A":"B"));
                await Measure("scroll",i=>editor.ScrollToVerticalOffset(view.DefaultLineHeight*(i%2==0?30:10)));
            }
            File.WriteAllText(Path.Combine(output,"results.json"),JsonSerializer.Serialize(new{fixture="LatexEditor, 1000 Chinese/TeX lines; plain vs 4 emoji per line; draw/input/scroll, 900x760 WPF, two warmups and seven samples; paired 80ms and idle",results},new JsonSerializerOptions{WriteIndented=true})); exit=0;
        } catch(Exception e) {Console.Error.WriteLine(e);} finally {window.Close();app.Shutdown(exit);} };
        window.Show(); app.Run(); return exit;
    }
}
